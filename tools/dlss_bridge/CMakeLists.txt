cmake_minimum_required(VERSION 3.20)
project(rsx_dlss_bridge LANGUAGES CXX)

# NGX's Windows static library uses the MSVC ABI. This optional C ABI DLL is
# built separately from the MinGW game runtime. No NVIDIA files are vendored.
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The optional NGX bridge requires 64-bit Windows MSVC")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The NGX bridge requires a 64-bit toolchain")
endif()
set(NGX_INCLUDE_DIR "" CACHE PATH "Official NVIDIA NGX headers")
set(NGX_EXTRA_INCLUDE_DIR "" CACHE PATH "Optional additional headers from the same SDK revision")
set(NGX_LIBRARY "" CACHE FILEPATH "Official MSVC /MD NGX library")
set(VULKAN_INCLUDE_DIR "" CACHE PATH "Directory containing vulkan/vulkan.h")
set(VULKAN_IMPORT_LIBRARY "" CACHE FILEPATH "Vulkan loader import library for the fixture")
foreach(required "${NGX_INCLUDE_DIR}/nvsdk_ngx_vk.h" "${NGX_LIBRARY}" "${VULKAN_INCLUDE_DIR}/vulkan/vulkan.h")
    if(NOT EXISTS "${required}")
        message(FATAL_ERROR "Missing explicit dependency: ${required}")
    endif()
endforeach()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SDK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(rsx_dlss_vk SHARED "${SDK_ROOT}/libs/video/rsx_dlss_vk.cpp")
target_include_directories(rsx_dlss_vk PUBLIC "${SDK_ROOT}/libs/video" "${VULKAN_INCLUDE_DIR}")
target_include_directories(rsx_dlss_vk PRIVATE "${NGX_INCLUDE_DIR}" "${NGX_EXTRA_INCLUDE_DIR}")
target_compile_options(rsx_dlss_vk PRIVATE /W4 /WX /EHsc)
target_link_libraries(rsx_dlss_vk PRIVATE "${NGX_LIBRARY}")

option(RSX_DLSS_BUILD_FIXTURE "Build the explicit synthetic GPU consumer test" OFF)
if(RSX_DLSS_BUILD_FIXTURE)
    if(NOT EXISTS "${VULKAN_IMPORT_LIBRARY}")
        message(FATAL_ERROR "Provide the Vulkan loader import library for the fixture")
    endif()
    foreach(fixture test_vulkan_dlss_consumer test_vulkan_dlss_fg_consumer)
        add_executable(${fixture} "${SDK_ROOT}/tests/${fixture}.cpp")
        target_include_directories(${fixture} PRIVATE "${NGX_INCLUDE_DIR}" "${NGX_EXTRA_INCLUDE_DIR}")
        target_compile_options(${fixture} PRIVATE /W4 /WX /EHsc)
        target_link_libraries(${fixture} PRIVATE rsx_dlss_vk "${NGX_LIBRARY}" "${VULKAN_IMPORT_LIBRARY}")
    endforeach()
endif()
